% The two answer chips, first chip centred at (#1,#2).
\newcommand{\chipsat}[2]{%
  \draw[line width=1.3pt,fill=white] (#1,#2) circle (0.85);
  \node[font=\bfseries\fontsize{20}{20}\selectfont] at (#1,#2) {1st};
  \draw[line width=1.3pt,fill=white] ({#1+2.3},#2) circle (0.85);
  \node[font=\bfseries\fontsize{20}{20}\selectfont] at ({#1+2.3},#2) {2nd};}

% One pile on its own line: numeral, counters (rows of 5), chips.
\newcommand{\kpile}[1]{%
  \begin{tikzpicture}
    \pgfmathtruncatemacro{\nrows}{(#1-1)/5+1}
    \pgfmathsetmacro{\midy}{-(\nrows-1)*1.5/2}
    \node[font=\fontsize{18}{18}\selectfont,text=black!65] at (0.35,\midy) {#1};
    \pileat{#1}{1.6}{0}{0.65}{1.5}
    \chipsat{14.4}{\midy}
    \path (0,0.9) (0,{\midy-0.9}) (0,{-(\nrows-1)*1.5-0.7});
  \end{tikzpicture}\par}

% A box holding a pile of n counters with its numeral.
% \kbox{n}{x}{y}{h}: top-left corner (x,y), height h; counters in rows of 5, centred vertically.
\newcommand{\kbox}[4]{%
  \pgfmathsetmacro{\bx}{#2}\pgfmathsetmacro{\by}{#3}
  \pgfmathtruncatemacro{\bnr}{(#1-1)/5+1}
  \pgfmathsetmacro{\bxx}{\bx+0.425}
  \pgfmathsetmacro{\byy}{\by-(#4+0.6)/2+(\bnr-1)*0.3}
  \draw[line width=1pt,rounded corners=3pt] (\bx,\by) rectangle ({\bx+3.25},{\by-#4});
  \node[anchor=north west,font=\fontsize{15}{15}\selectfont,inner sep=2pt] at ({\bx+0.08},{\by-0.06}) {#1};
  \pileat{#1}{\bxx}{\byy}{0.27}{0.6}}

% Boxes for piles 1..N in rows of 5: \kboxes{N}{box height}
\newcommand{\kboxes}[2]{%
  \begin{tikzpicture}
    \foreach \n in {1,...,#1} {
      \pgfmathtruncatemacro{\brow}{(\n-1)/5}
      \pgfmathtruncatemacro{\bcol}{mod(\n-1,5)}
      \kbox{\n}{\bcol*3.72}{-\brow*(#2+0.42)}{#2}
    }
  \end{tikzpicture}\par}

% A light tray behind a row of n counters, first counter centre (x,y), pitch p, radius r
\newcommand{\tray}[5]{%
  \fill[black!9,rounded corners=8pt] ({#2-#5-0.18},{#3-#5-0.16}) rectangle ({#2+(#1-1)*#4+#5+0.18},{#3+#5+0.16});}

% A case made of rows (one row of counters per entry), with chips at the right.
% \krows{list of row lengths}{number of rows}
\newcommand{\krows}[2]{%
  \begin{tikzpicture}
    \pgfmathsetmacro{\midy}{-(#2-1)*1.9/2}
    \draw[black!35,line width=0.8pt,rounded corners=6pt] (0,1.05) rectangle (\linewidth-1pt,{-(#2-1)*1.9-1.05});
    \foreach \len [count=\r] in {#1} {
      \tray{\len}{1.6}{-(\r-1)*1.9}{1.42}{0.65}
      \foreach \c in {1,...,\len} {
        \cnt{({0.18+\c*1.42},{-(\r-1)*1.9})}{0.65}
      }
    }
    \chipsat{14.1}{\midy}
  \end{tikzpicture}\par}

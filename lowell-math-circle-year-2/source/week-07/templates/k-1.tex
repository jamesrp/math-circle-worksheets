% Rows of piles, each followed by "1st" and "2nd" to circle.
\newcommand{\rowstep}{0.72}
\newcommand{\pilechoice}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \n [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*\rowstep}
      \foreach \k in {1,...,\n}{\counter{(\k-1)*0.55+0.3}{\y}}
      \node[font=\LARGE] at (5.3,\y) {1st};
      \node[font=\LARGE] at (6.7,\y) {2nd};
    }
  \end{tikzpicture}}

% Rows of piles with room to circle counters.
\newcommand{\pilerows}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \n [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*0.76}
      \foreach \k in {1,...,\n}{\counter{(\k-1)*0.66+0.3}{\y}}
    }
  \end{tikzpicture}}

% Two piles side by side, then "1st" and "2nd" to circle.
\newcommand{\twopilechoice}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \a/\b [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*\rowstep}
      \foreach \k in {1,...,\a}{\counter{(\k-1)*0.48+0.3}{\y}}
      \foreach \k in {1,...,\b}{\counter{(\k-1)*0.48+2.95}{\y}}
      \node[font=\LARGE] at (5.3,\y) {1st};
      \node[font=\LARGE] at (6.7,\y) {2nd};
    }
  \end{tikzpicture}}

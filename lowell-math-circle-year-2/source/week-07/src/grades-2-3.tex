% Rows of piles, each followed by "first" and "second" to circle.
% Optional argument: distance between counters, in inches.
\newcommand{\pilechoice}[2][0.52]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \n [count=\i] in {#2}{%
      \pgfmathsetmacro{\y}{-(\i-1)*0.66}
      \foreach \k in {1,...,\n}{\counter{(\k-1)*#1+0.3}{\y}}
      \node[font=\Large] at (5.4,\y) {first};
      \node[font=\Large] at (6.75,\y) {second};
    }
  \end{tikzpicture}}

% Rows of two piles side by side.
\newcommand{\twopiles}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \a/\b [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*0.7}
      \foreach \k in {1,...,\a}{\counter{(\k-1)*0.52+0.3}{\y}}
      \foreach \k in {1,...,\b}{\counter{(\k-1)*0.52+3.8}{\y}}
    }
  \end{tikzpicture}}

\begin{document}

Two players take turns. Whoever takes the last counter wins. On a track,
the token starts on the number of counters and moves toward 0 by the
number of counters taken, and whoever puts it on 0 wins.

\vspace{0.25in}

\begin{problem}
Play each pile with your partner, taking 1, 3 or 4 counters on each turn,
until you are sure whether you would rather go first or second. Circle
your choice. If you would rather go first, also circle the counters you
take on your first turn.

\vspace{0.35in}
\pilechoice{2,5,6,7,8}
\end{problem}

\begin{problem}
The token moves 1, 3 or 4 squares. Colour every square where you would
like to leave the token for your opponent.

\vspace{0.2in}
\track{20}\gap\track{20}
\end{problem}

\begin{problem}
You and your partner play the adult together, moving the token 1, 3 or 4
squares. The adult puts the token on a square from 10 to 20, and you
choose who goes first. For each game, circle the start and colour each
square where you put the token.

\vspace{0.2in}
\track{20}\gap\track{20}\gap\track{20}
\end{problem}

\begin{problem}
The token starts on 9 and moves 1, 3 or 4 squares. Lena says that if her
opponent goes first, she can win whatever her opponent does. Is she
right? Explain.

\vspace{0.2in}
\track{10}
\vspace{1.6in}
\end{problem}

\newpage

\begin{problem}
Omar plays with moves of 1, 3 or 4, and he always takes as many counters
as he is allowed. He goes first with a pile of 13. Does Omar win whatever
his opponent does? Explain. Then find a way to play from 13 that wins
whatever the opponent does.

\vspace{0.2in}
\track{20}
\vspace{1.8in}
\end{problem}

\begin{problem}
With moves of 1, 3 or 4, would you rather go first or second with a pile
of 50? With a pile of 100? If you would go first, write how many counters
you would take. Explain how you know.

\vspace{0.2in}
\track{30}
\vspace{1.5in}
\end{problem}

\newpage

\begin{problem}
Now make two piles. On each turn, take 1, 3 or 4 counters from one of the
piles. Whoever takes the last counter on the table wins. With piles of 5
and 5, and with piles of 5 and 6, would you rather go first or second?
Explain why your way of playing wins whatever your opponent does.

\vspace{0.4in}
\twopiles{5/5,5/6}
\vspace{3.2in}
\end{problem}

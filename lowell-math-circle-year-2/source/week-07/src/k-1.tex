% Rows of piles, each followed by "1st" and "2nd" to circle.
\newcommand{\rowstep}{0.72}
\newcommand{\pilechoice}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \n [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*\rowstep}
      \foreach \k in {1,...,\n}{\counter{(\k-1)*0.55+0.3}{\y}}
      \node[font=\LARGE] at (5.3,\y) {1st};
      \node[font=\LARGE] at (6.7,\y) {2nd};
    }
  \end{tikzpicture}}

% Rows of piles with room to circle counters.
\newcommand{\pilerows}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \n [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*0.76}
      \foreach \k in {1,...,\n}{\counter{(\k-1)*0.66+0.3}{\y}}
    }
  \end{tikzpicture}}

% Two piles side by side, then "1st" and "2nd" to circle.
\newcommand{\twopilechoice}[1]{%
  \begin{tikzpicture}[x=1in,y=1in]
    \foreach \a/\b [count=\i] in {#1}{%
      \pgfmathsetmacro{\y}{-(\i-1)*\rowstep}
      \foreach \k in {1,...,\a}{\counter{(\k-1)*0.48+0.3}{\y}}
      \foreach \k in {1,...,\b}{\counter{(\k-1)*0.48+2.95}{\y}}
      \node[font=\LARGE] at (5.3,\y) {1st};
      \node[font=\LARGE] at (6.7,\y) {2nd};
    }
  \end{tikzpicture}}

\newcommand{\gap}{\par\vspace{0.2in}}
\newcommand{\threetracks}{\track{10}\gap\track{10}\gap\track{10}\par}
\newcommand{\twotracks}{\track{10}\gap\track{10}\par}
% A track with "1st" and "2nd" on their own line, well below it.
\newcommand{\choicetrack}{%
  \track{10}\par\vspace{0.3in}%
  \hspace*{0.35in}{\LARGE 1st}\hspace{1.0in}{\LARGE 2nd}\par}
\newcommand{\gamelabel}[1]{\par\vspace{0.14in}#1\par\vspace{0.1in}}

\begin{document}

Two players take turns. On your turn, take 1 or 2 counters, or move the
token 1 or 2 squares toward 0. Whoever takes the last counter, or puts the
token on 0, wins.

\vspace{0.35in}

\begin{problem}
Make each pile and play it many times with your partner. Circle whether
you would rather go 1st or 2nd.

\vspace{0.4in}
\pilechoice{2,3,4,5,6,7,8}
\end{problem}

\newpage

\begin{problem}
With your partner, play the grown-up from 10. The grown-up goes first, and
you colour the grown-up's squares.

\vspace{0.2in}
\threetracks
\end{problem}

\begin{problem}
Now you go first from 10 against your partner. Colour your own squares.

\vspace{0.2in}
\threetracks
\end{problem}

\newpage

\begin{problem}
You go first against your partner with each pile. Circle the counters you
take first so that you can win.

\vspace{0.4in}
\pilerows{5,7,8,10}
\end{problem}

\begin{problem}
Your partner puts the token on a square from 1 to 10. Circle it, and
circle 1st or 2nd so that you can win.

\vspace{0.2in}
\choicetrack\vspace{0.25in}
\choicetrack\vspace{0.25in}
\choicetrack
\end{problem}

\newpage

\begin{problem}
Now move the token 1, 2 or 3 squares, starting on 10. Colour the first
player's squares.

\gamelabel{Play the grown-up, who goes first.}
\twotracks
\gamelabel{Play your partner, and you go first.}
\threetracks
\end{problem}

\newpage

\begin{problem}
Now move the token 1 or 4 squares, starting on 8. Colour the first
player's squares.

\gamelabel{Play the grown-up, who goes first.}
\twotracks
\gamelabel{Play your partner, and you go first.}
\threetracks
\end{problem}

\newpage
\renewcommand{\rowstep}{0.66}

\begin{problem}
Now make two piles, and take 1 or 2 counters from one pile. Whoever takes
the last counter on the table wins.

\vspace{0.32in}
\twopilechoice{2/2,1/2,3/3,2/4,4/4}
\end{problem}

\begin{problem}
Now whoever takes the last counter loses. Play each pile many times with
your partner.

\vspace{0.32in}
\pilechoice{1,2,3,4,5,6}
\end{problem}

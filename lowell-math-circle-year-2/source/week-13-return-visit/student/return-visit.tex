% Full-size working maps. Coordinates are centimetres, never independently scaled.
\newcommand{\DotMap}[1]{%
  \begin{tikzpicture}[x=1cm,y=1cm]
    \path[use as bounding box] (-.40,-3.10) rectangle (16.45,3.10);
    \coordinate (S) at (0,0); \coordinate (A) at (3.7,1.7);
    \coordinate (B) at (3.7,-1.7); \coordinate (H) at (8,0);
    \coordinate (C) at (12.3,1.7); \coordinate (D) at (12.3,-1.7);
    \coordinate (T) at (16,0);
    \foreach \u/\v in {S/A,S/B,A/H,B/H,H/C,H/D,C/T,D/T}
      \draw[road] (\u)--(\v);
    \ifnum#1=1 \draw[road] (A)--(C); \fi
    \foreach \p in {S,A,B,H,C,D,T} \node[point] at (\p) {};
    \node[letter] at (0,.80) {S}; \node[letter] at (16,.80) {T};
    % Nearest label-box edge is 9.5 mm from each reservable dot.
    % A 15 mm diameter marker has radius 7.5 mm; visible ink clears it.
    \node[letter,anchor=south] at (3.7,2.65) {A};
    \node[letter,anchor=north] at (3.7,-2.65) {B};
    \node[letter,anchor=south] at (8,.95) {H};
    \node[letter,anchor=south] at (12.3,2.65) {C};
    \node[letter,anchor=north] at (12.3,-2.65) {D};
  \end{tikzpicture}%
}

\newcommand{\DotExample}[2]{%
  \begin{tikzpicture}[x=1cm,y=1cm]
    \path[use as bounding box] (-.3,-1.10) rectangle (4.6,1.05);
    \coordinate (U) at (0,0); \coordinate (V) at (2.1,0);
    \coordinate (W) at (4.2,0);
    \ifnum#1=2 \fill[gray!35] (V) circle (.75cm); \fi
    \draw[road] (U)--(V); \draw[road] (V)--(W);
    \ifnum#1>0
      \draw[line width=1.6pt,densely dashed] (U)--(V)--(W);
    \fi
    \foreach \p in {U,V,W} \node[point] at (\p) {};
    \node[letter] at (0,-.42) {U};
    \node[letter] at (2.1,-.97) {V};
    \node[letter] at (4.2,-.42) {W};
    \node[font=\small] at (2.1,.94) {#2};
  \end{tikzpicture}%
}

\newcommand{\CapacityMap}[1]{%
  \begin{tikzpicture}[x=1cm,y=1cm]
    \path[use as bounding box] (-.40,-2.45) rectangle (16.45,2.45);
    \coordinate (S) at (0,0); \coordinate (A) at (5,1.7);
    \coordinate (B) at (5,-1.7); \coordinate (C) at (10.8,0);
    \coordinate (T) at (16,0);
    \draw[road] (S)--node[capacity,above=3pt]{3} (A);
    \draw[road] (S)--node[capacity,below=3pt]{2} (B);
    \draw[road] (A)--node[capacity,below=3pt]{2} (C);
    \draw[road] (B)--node[capacity,above=3pt]{2} (C);
    \draw[road] (C)--node[capacity,below=5pt]{#1} (T);
    \draw[road] (A)--node[capacity,above=5pt]{1} (T);
    \foreach \p in {S,A,B,C,T} \node[point] at (\p) {};
    \node[letter] at (0,.80) {S}; \node[letter] at (16,-.80) {T};
    \node[letter] at (5,2.36) {A}; \node[letter] at (5,-2.36) {B};
    \node[letter] at (10.8,-.80) {C};
  \end{tikzpicture}%
}

\newcommand{\CapacityExample}[2]{%

    \foreach \p/\x in {U/0,V/2.2,W/4.4}

    \node[font=\small] at (2.2,.85) {#2};
    \ifnum#1>0 \node[anchor=west,font=\normalsize] at (.35,-.82) {1: U V W}; \fi
    \ifnum#1=2 \node[anchor=west,font=\normalsize] at (.35,-1.28) {2: U V W}; \fi
  \end{tikzpicture}%
}

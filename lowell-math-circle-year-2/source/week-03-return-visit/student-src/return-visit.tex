% Compact card orders are records; the five-slot mat and supplied cards are 25 mm squares.
\newcommand{\cardrow}[4]{%
 \foreach \val [count=\i from 0] in {#4}{\pgfmathsetmacro{\xx}{#1+\i*#3}\draw[ink,line width=.5pt] (\xx,#2) rectangle ({\xx+#3-.09},{#2-.75});\node at ({\xx+(#3-.09)/2},{#2-.375}) {\val};}
}
\newcommand{\recordlines}[3]{\foreach \k in {0,...,#3}{\draw[rulegray,line width=.45pt] (#1,#2-.58*\k)--(#1+17.55,#2-.58*\k);}}
% Empty arrow-machine records. Each endpoint label is a slot number, never a card value.
\newcommand{\blankmachine}[4]{%
 \node[anchor=west,font=\sffamily\fontsize{10}{12}\selectfont] at (#1,#2+1.18) {After two passes: #4};
 \foreach \i in {1,...,#3}{\pgfmathsetmacro{\xx}{#1+(\i-.5)*1.35}\draw[ink,line width=.55pt] (\xx-.48,#2+.32) rectangle (\xx+.48,#2-.32);\draw[ink,line width=.55pt] (\xx-.48,#2-1.65) rectangle (\xx+.48,#2-2.29);\node[font=\sffamily\fontsize{9}{11}\selectfont] at (\xx,#2+.58) {\i};\node[font=\sffamily\fontsize{9}{11}\selectfont] at (\xx,#2-2.55) {\i};}
 \node[anchor=east,font=\sffamily\fontsize{8.5}{10}\selectfont] at (#1-.05,#2) {in};
 \node[anchor=east,font=\sffamily\fontsize{8.5}{10}\selectfont] at (#1-.05,#2-1.97) {out};
}

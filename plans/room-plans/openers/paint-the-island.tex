\newcommand{\mainmap}[1]{\begin{tikzpicture}[x=#1cm,y=#1cm]
  \draw[border] (0,0) rectangle (4,3);
  \draw[border] (0,2) -- (3,2) (2,2) -- (2,3) (1,0) -- (1,2) (2,1) -- (2,2) (2,1) -- (4,1) (3,0) -- (3,2);
  \foreach \p/\n in {{(1,2.5)}/Ant,{(3,2.5)}/Bear,{(0.5,1)}/Cat,{(1.5,1)}/Dog,{(2.5,1.5)}/Eel,{(3.5,0.5)}/Fox}
    \node[font=\large\bfseries] at \p {\n};
  \end{tikzpicture}}
\newcommand{\smallmap}[1]{\begin{tikzpicture}[x=#1cm,y=#1cm]
  \draw[border] (0,0) rectangle (2,2);
  \draw[border] (1,0) -- (1,2) (0,1) -- (2,1);
  \foreach \p/\n in {{(0.5,1.5)}/Ant,{(1.5,1.5)}/Bear,{(0.5,0.5)}/Cat,{(1.5,0.5)}/Dog}
    \node[font=\large\bfseries] at \p {\n};
  \end{tikzpicture}}

\newcommand{\cutline}{\par\noindent\begin{tikzpicture}\draw[dashed,gray] (0,0)--(\linewidth,0);\end{tikzpicture}\par\vspace{0.3in}}
\newcommand{\heading}[1]{{\LARGE\bfseries #1}\par\medskip}
\newcommand{\twocol}[2]{\begin{minipage}[t]{0.50\linewidth}\vspace{0pt}#1\end{minipage}\hfill
  \begin{minipage}[t]{0.46\linewidth}\vspace{0pt}\centering #2\end{minipage}}
\newcommand{\therule}{Every country on the island is red, blue or yellow. Two countries that share a border
  are different colors. Countries that touch only at a corner may match.}
\newcommand{\mainpuzzle}{%
  \heading{Paint the island}
  \therule\ Color each country, or write its color in it.\par
  \medskip
  \twocol{%
  \begin{enumerate}\setlength{\itemsep}{5pt}
  \item Ant Land is red.
  \item Ant Land and Fox Land are different colors.
  \item Cat Land and Eel Land are the same color.
  \item Exactly two countries are blue.
  \end{enumerate}\par
  \medskip
  \textbf{Done?} Cross out clue 4. How many answers are there now? Write a new clue that makes it exactly one again.}{\mainmap{2.0}}}

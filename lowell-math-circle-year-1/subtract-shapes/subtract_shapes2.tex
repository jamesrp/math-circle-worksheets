% Define a command for drawing a red square with content
\newcommand{\redbox}[1]{%
  \begin{tikzpicture}[scale=0.8, baseline=(current bounding box.center)]
    % Red square border
    \draw[red, ultra thick] (0,0) rectangle (2,2);
    % Content
    #1
  \end{tikzpicture}%
}

% Redbox only
\newcommand{\redboxonly}{%
  \begin{tikzpicture}[scale=0.8, baseline=(current bounding box.center)]
    % Red square border
    \draw[red, ultra thick] (0,0) rectangle (2,2);
  \end{tikzpicture}%
}

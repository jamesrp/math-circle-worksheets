\documentclass{article}
\usepackage{tikz}
\usepackage{amsmath}

\begin{document}

% Define a command for drawing a red square with content
\newcommand{\redbox}[1]{%
  \begin{tikzpicture}[scale=0.8]
    % Red square border
    \draw[red, thick] (0,0) rectangle (2,2);
    % Content
    #1
  \end{tikzpicture}%
}

% Define individual shapes
\newcommand{\diagonal}{\draw[black, thick] (0,0) -- (2,2);}
\newcommand{\antidiagonal}{\draw[black, thick] (0,2) -- (2,0);}
\newcommand{\horizontal}{\draw[black, thick] (0,1) -- (2,1);}
\newcommand{\vertical}{\draw[black, thick] (1,0) -- (1,2);}
\newcommand{\topright}{\draw[black, thick] (1,1) -- (2,2);}
\newcommand{\topleft}{\draw[black, thick] (1,1) -- (0,2);}
\newcommand{\bottomright}{\draw[black, thick] (1,1) -- (2,0);}
\newcommand{\bottomleft}{\draw[black, thick] (1,1) -- (0,0);}
\newcommand{\lshape}{\draw[black, thick] (0,0) -- (0,1) -- (1,1);}
\newcommand{\corner}{\draw[black, thick] (1,1) -- (2,1) -- (2,0);}

\begin{center}

% Problem 1: Diagonal + Anti-diagonal = X
\redbox{\diagonal} 
$+$ 
\redbox{\antidiagonal} 
$=$ 
\redbox{\diagonal \antidiagonal}

\vspace{1cm}

% Problem 2: Horizontal + Vertical = Plus
\redbox{\horizontal} 
$+$ 
\redbox{\vertical} 
$=$ 
\redbox{\horizontal \vertical}

\vspace{1cm}

% Problem 3: Top-right + Bottom-left = Diagonal
\redbox{\topright} 
$+$ 
\redbox{\bottomleft} 
$=$ 
\redbox{\diagonal}

\vspace{1cm}

% Problem 4: L-shape + Corner = Square frame
\redbox{\lshape} 
$+$ 
\redbox{\corner} 
$=$ 
\redbox{
  \draw[black, thick] (0,0) -- (0,1) -- (1,1);
  \draw[black, thick] (1,1) -- (2,1) -- (2,0);
}

\vspace{1cm}

% Problem 5: All four center lines = Star
\redbox{\topright} 
$+$ 
\redbox{\topleft \bottomright} 
$=$ 
\redbox{\topright \topleft \bottomright}

\vspace{1cm}

% Problem 6: Horizontal + Diagonal = Cross pattern
\redbox{\horizontal} 
$+$ 
\redbox{\diagonal} 
$=$ 
\redbox{\horizontal \diagonal}

\vspace{1cm}

% Subtraction Problems

% Problem 7: X minus diagonal = anti-diagonal
\redbox{\diagonal \antidiagonal} 
$-$ 
\redbox{\diagonal} 
$=$ 
\redbox{\antidiagonal}

\vspace{1cm}

% Problem 8: Plus minus vertical = horizontal
\redbox{\horizontal \vertical} 
$-$ 
\redbox{\vertical} 
$=$ 
\redbox{\horizontal}

\vspace{1cm}

% Problem 9: Star pattern minus one line
\redbox{
  \topright \topleft \bottomright \bottomleft
} 
$-$ 
\redbox{\bottomleft} 
$=$ 
\redbox{\topright \topleft \bottomright}

\end{center}

\end{document}
